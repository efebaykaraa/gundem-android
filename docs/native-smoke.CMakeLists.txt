# Standalone host smoke: see README for invocation.
cmake_minimum_required(VERSION 3.22)
project(gundem_smoke C CXX)
set(CMAKE_CXX_STANDARD 17)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
set(LLAMA_OPENSSL OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_SERVER OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_COMMON OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TOOLS OFF CACHE BOOL "" FORCE)
add_subdirectory(${GUNDEM_ROOT}/vendor/llama.cpp llama)
add_executable(gundem-smoke ${GUNDEM_ROOT}/docs/native-smoke.cpp ${GUNDEM_ROOT}/app/src/main/cpp/engine.cpp)
target_include_directories(gundem-smoke PRIVATE ${GUNDEM_ROOT}/app/src/main/cpp)
target_link_libraries(gundem-smoke PRIVATE llama)
